export DATA_FOLDER=`pwd`/base
mkdir -p ${DATA_FOLDER}

# download data to run the demo
docker run --rm \
  -v "${DATA_FOLDER}:/opt/data" \
  --user "$(id -u):$(id -g)" \
  miltondp/phenoplier \
  /bin/bash -c "python environment/scripts/setup_data.py --mode demo"

export CODE_FOLDER=`pwd`
cd

docker run --rm \
  -p 8888:8892 \
  -v "${DATA_FOLDER}:/opt/data" \
  -v "${CODE_FOLDER}:/opt/code" \
  --user "$(id -u):$(id -g)" \
  miltondp/phenoplier
